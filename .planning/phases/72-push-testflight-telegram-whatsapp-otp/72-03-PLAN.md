- export type PhoneCountry = { iso: string; name: string; flag: string; dialCode: string; nationalDigits: number; groups: number[] }
- export const PHONE_COUNTRIES: PhoneCountry[] в порядке: KZ "Казахстан" dial "7" 10 цифр groups [3,3,2,2]; RU "Россия" "7" 10 [3,3,2,2]; UZ "Узбекистан" "998" 9 [2,3,2,2]; KG "Кыргызстан" "996" 9 [3,3,3]; TJ "Таджикистан" "992" 9 [2,3,2,2]; AZ "Азербайджан" "994" 9 [2,3,2,2]; GE "Грузия" "995" 9 [3,3,3]; BY "Беларусь" "375" 9 [2,3,2,2]; AM "Армения" "374" 8 [2,3,3]; MD "Молдова" "373" 8 [2,3,3]; TM "Туркменистан" "993" 8 [2,3,3]. flag — эмодзи флага страны. - export const DEFAULT_PHONE_COUNTRY = PHONE_COUNTRIES[0] (KZ, сохраняет текущее поведение).
- export function extractNationalDigits(country, rawInput): только цифры; если строка ввода начинается с "+" и цифры начинаются с dialCode — отрезать dialCode; иначе для dialCode "7" отрезать одну ведущую 7 или 8 (как сейчас); обрезать до nationalDigits.
- export function formatNationalNumber(country, digits): `+${dialCode} ` + цифры, разбитые по groups и соединённые пробелами (пустые группы не выводятся).
- export function toE164(country, digits): `+${dialCode}${digits}`.
- export function isCompleteNationalNumber(country, digits): digits.length === country.nationalDigits.
